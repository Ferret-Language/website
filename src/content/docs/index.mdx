  <div class="scroll-fade-in" data-delay="200">
  
```ferret
type Mode enum {
  debug,
  run,
}

fn SelectMode(v: i32) -> Mode {
  if v > 0 {
    return Mode::run
  }
  return Mode::debug
}

fn main() {
  let mode = SelectMode(1)
  if mode == Mode::run {
    println("Ferret is running")
  }
}
```
