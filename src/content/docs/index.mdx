  <div class="section-label scroll-fade-in">Quick Look</div>
  <div class="scroll-fade-in" data-delay="200">
  
```ferret
import "std/io"

type Mode enum {
  debug,
  run,
}

fn SelectMode(v i32) Mode {
  if v > 0 {
    return Mode::run
  }
  return Mode::debug
}

fn main() i32 {
  let mode = SelectMode(1)
  if mode == Mode::run {
    io::Println("Ferret is running")
  }
  return 0
}
```
